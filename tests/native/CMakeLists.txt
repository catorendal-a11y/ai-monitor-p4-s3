cmake_minimum_required(VERSION 3.20)
project(ai_monitor_ui_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "../.." ABSOLUTE)
set(LVGL_ROOT "${PROJECT_ROOT}/.pio/libdeps/esp32p4-release/lvgl")
if(NOT EXISTS "${LVGL_ROOT}/lvgl.h")
  message(FATAL_ERROR "Build esp32p4-release first to install the pinned LVGL sources")
endif()
# Use the same pinned LVGL code and lv_conf.h as the firmware, without window
# drivers, network dependencies, or an attached screen.
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_ROOT}/src/*.c")
add_library(lvgl_native STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_native PUBLIC "${LVGL_ROOT}" "${PROJECT_ROOT}/lib")
target_compile_definitions(lvgl_native PUBLIC LV_CONF_INCLUDE_SIMPLE=1 LV_USE_LZ4_INTERNAL=1)
add_executable(test_ui test_ui.cpp "${PROJECT_ROOT}/src/ui/idle_dim.cpp" "${PROJECT_ROOT}/src/ui/nova_assets.cpp" "${PROJECT_ROOT}/src/companions/builtin_lz4.cpp")
target_include_directories(test_ui PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_ROOT}/src")
target_compile_definitions(test_ui PRIVATE AIM_COMPANION_CACHE_TEST=1)
target_link_libraries(test_ui PRIVATE lvgl_native)
if(NOT MSVC)
  target_compile_options(test_ui PRIVATE -Wall -Wextra -Werror=return-type)
endif()
enable_testing()
add_test(NAME ui_regressions COMMAND test_ui "${CMAKE_CURRENT_BINARY_DIR}/ctest-previews")

add_executable(test_ui_s3 test_ui.cpp "${PROJECT_ROOT}/src/ui/idle_dim.cpp" "${PROJECT_ROOT}/src/ui/nova_assets.cpp" "${PROJECT_ROOT}/src/companions/builtin_lz4.cpp")
target_include_directories(test_ui_s3 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_ROOT}/src")
target_compile_definitions(test_ui_s3 PRIVATE AIM_BOARD_WAVESHARE_S3=1 AIM_COMPANION_CACHE_TEST=1)
target_link_libraries(test_ui_s3 PRIVATE lvgl_native)
add_test(NAME s3_ui_regressions COMMAND test_ui_s3 "${CMAKE_CURRENT_BINARY_DIR}/ctest-previews/s3")

# Exercise the actual permission-screen wrapper with bounded local publication fixtures.
# ESP/serial/IPC are not opened by this target; transport crypto has separate native tests.
add_executable(test_ui_permissions test_ui.cpp "${PROJECT_ROOT}/src/ui/idle_dim.cpp" "${PROJECT_ROOT}/src/ui/nova_assets.cpp" "${PROJECT_ROOT}/src/companions/builtin_lz4.cpp")
target_include_directories(test_ui_permissions PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_ROOT}/src" "${PROJECT_ROOT}/.pio/libdeps/esp32p4-release/ArduinoJson/src")
target_compile_definitions(test_ui_permissions PRIVATE AIM_USB_PERMISSION_RUNTIME=1 AIM_COMPANION_CACHE_TEST=1)
target_link_libraries(test_ui_permissions PRIVATE lvgl_native)
add_test(NAME permission_wrapper_regressions COMMAND test_ui_permissions "${CMAKE_CURRENT_BINARY_DIR}/ctest-previews/permission-wrapper")

add_executable(test_mirror test_mirror.cpp)
target_include_directories(test_mirror PRIVATE "${PROJECT_ROOT}/.pio/libdeps/esp32p4-release/ArduinoJson/src")
target_link_libraries(test_mirror PRIVATE lvgl_native)
if(NOT MSVC)
  target_compile_options(test_mirror PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME mirror_regressions COMMAND test_mirror)
